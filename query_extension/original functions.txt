procedure RVExtension(toArma: PAnsiChar; outputSize: Integer; fromArma: PAnsiChar); stdcall; export;
var
  FunctionName: string;
  LineString, ParamString: string;
  Params: TStringList;
begin
  FunctionName := copy(string(fromArma), 1, pos(':', string(fromArma)) - 1);
  LineString := copy(string(fromArma), length(FunctionName) + 2, length(string(fromArma)));
  FunctionName := FunctionName + ':' + copy(string(LineString), 1, pos(':', string(LineString)) - 1);
  LineString := copy(string(fromArma), length(FunctionName) + 2, length(string(fromArma)));
  Params := TStringList.Create;
  try
    Params.Clear;
    repeat
      if pos(':', LineString) > 0 then
        ParamString := copy(LineString, 1, pos(':', LineString) - 1)
      else
        ParamString := LineString;
      LineString := copy(LineString, pos(':', LineString) - 1, length(ParamString));
      Params.Add(ParamString);
      ParamString := EmptyStr;
    until (LineString = EmptyStr) or (pos(':', LineString) = 0);
    SetLength(LineString,outputSize);
    LineString := ExecuteFunction(FunctionName, Params);
    StrPCopy(toArma, LineString);
  finally
    FreeAndNil(Params);
  end;
end;

function TDataModuleMySQL.ExecuteSelect(TheParams: TStrings): string;
var
  I: integer;
begin
  Result := EmptyStr;
  try
    try
      with mySQLQuery, mySQLQuery.SQL do
        begin
          Clear;
          AddStrings(TheParams);
          Open;
          while not EOF do
            begin
              for I := 0 to FieldCount - 1 do
                begin
                  Result := Result + Fields[I].asString;
                  Result := Result + '|';
                end;
              Result := Result + '#EOL#';
              Next;
            end; //while not EOF do
          Result := Result + '#EOF#';
        end; //with mySQLQuery, mySQLQuery.SQL do
    except
      on E: exception do
        Result := E.Message;
    end;
  finally
    if mySQLQuery.Active then
      mySQLQuery.Close;
  end;
end;
